▸3처음에는 범위와 파일 수명을 한곳에서 관리하는 영역 방식이 개발하기 더 단순하겠다고 생각했습니다.

▸4다만 영역 정보를 새로 만들면 등록, 실패 시 정리, 해제 경로까지 SPT와 함께 관리해야 했습니다.

for (uintptr_t upage = (uintptr_t)pg_round_down(buffer); upage <= end;
     upage += PGSIZE) {
    struct page *page;
    if ((page = spt_find_page(&thread_current()->spt, (void *)upage))) {
        if (page->frame != NULL) {
            continue;
        }
        if (!vm_claim_page((void *)upage)) {
            return false;
        }
    } else {
        if (!stack_growth((void *)upage,

            return false;
        }
    }
}
